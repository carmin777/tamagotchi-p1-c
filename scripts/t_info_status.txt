# t_hatch.txt - boot, set 08:00, go to the egg, wait for hatching and leave the baby alone for 20 min.
wait 4
press M           # clock setting
wait 2.3
press L
press L
press L
press L
press L
press L
press L
press L           # 08:00 AM
press R           # confirm
wait 3.3
press M           # pet screen (egg)
mark egg
wait 330.3        # baby hatched
press L
press L
press L
press L
press L
press L           # icon 6: status
mark start
press M           # page 0: age/weight
wait 1.2
dump p0
press L           # page 1: discipline
wait 1.3
press M           # page 2: hunger
wait 1.4
press L           # page 3: happiness
wait 1.1
press M           # back to page 0
wait 1.5
press R           # exits
wait 2.2
dump exit
press M           # enters the status again (cursor kept)
wait 2.1
press L
wait 17           # 15 s timeout
dump timeout
mark end
wait 3
