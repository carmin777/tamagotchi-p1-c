# boot_set_clock.txt - powers on the device and sets the clock to 08:00 AM, then waits for the egg.
# Measured flow (see re/flows.md): egg/start screen -> M enters the clock setting;
# L = +1 hour (24h, shows AM/PM); M = +1 minute; R = confirm (the clock starts running);
# M again = leaves the setting and shows the egg.
wait 4            # boot: ~0.4s until the first screen; wait for the initial animation
press M           # enters the clock setting
wait 2            # the transition eats the first button if we do not wait
mark set_hour
press L           # 1
press L           # 2
press L           # 3
press L           # 4
press L           # 5
press L           # 6
press L           # 7
press L           # 8   -> 08:00 AM
mark confirm
press R           # confirms the time
wait 10
mark egg
press M           # exits to the egg
wait 6
dump boot_egg
save out/boot_egg.sav
